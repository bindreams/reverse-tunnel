# reverse-tunnel-frpc
[Unit]
Description=reverse-tunnel-frpc
StartLimitIntervalSec=0

[Service]
Type=exec
ExecStart=<ROOT>/bin/frpc -c ../host/frpc.toml
WorkingDirectory=<ROOT>/logs
Environment=RT_LABEL=alpha
Environment=RT_PROTOCOL=quic
Environment=RT_SERVER_ADDR=bridge.example.com
Environment=RT_SERVER_PORT=7000
Environment=RT_WS_PORT=8022
User=svc-example
Restart=always
RestartSec=5
StandardOutput=append:<ROOT>/logs/frpc.log
StandardError=append:<ROOT>/logs/frpc.log

[Install]
WantedBy=multi-user.target
